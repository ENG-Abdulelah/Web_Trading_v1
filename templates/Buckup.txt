async function api(url, method = 'GET', body = null) {
    const opts = { method, headers: { 'Content-Type': 'application/json' } };
    if (body) opts.body = JSON.stringify(body);
    const r = await fetch(url, opts);
    return r.json();
}

async function approveSignal(uid) {
    if (!confirm('{{ t.approve }}?')) return;
    const res = await api('/api/approve/' + uid, 'POST');
    if (res.ok) { alert('OK: ' + res.ticket); location.reload(); }
    else { alert('Error: ' + (res.error || '')); }
}

async function rejectSignal(uid) {
    await api('/api/reject/' + uid, 'POST');
    location.reload();
}

async function runAnalysis() {
    const btn = event.target.closest('button');
    const orig = btn.innerHTML;
    btn.disabled = true;
    btn.innerHTML = '<i class="bi bi-hourglass-split"></i>';
    const res = await api('/api/run-analysis', 'POST');
    alert(`Generated: ${res.generated}\nRejected: ${res.rejected}\nRaw: ${res.raw_setups || 0}`);
    location.reload();
}

async function refreshNews() {
    const res = await api('/api/news/refresh', 'POST');
    if (res.ok) { alert(`+${res.count}`); location.reload(); }
    else { alert('News fetch failed'); }
}

// --- Timeframe Filtering Logic ---
function updateTimeframeChips() {
    const style = document.getElementById('set-style').value;
    const swingTfs = ['1H', '4H'];
    const scalpTfs = ['1M', '5M', '15M'];

    document.querySelectorAll('.tf-chip').forEach(chip => {
        const tf = chip.dataset.tf;
        let show = false;

        if (style === 'swing' && swingTfs.includes(tf)) show = true;
        else if (style === 'scalping' && scalpTfs.includes(tf)) show = true;
        else if (style === 'both' && [...swingTfs, ...scalpTfs].includes(tf)) show = true;

        if (show) {
            chip.style.display = 'inline-block';
        } else {
            chip.style.display = 'none';
            chip.classList.remove('active');
        }
    });
}

document.getElementById('set-style').addEventListener('change', updateTimeframeChips);
document.addEventListener('DOMContentLoaded', updateTimeframeChips);

document.querySelectorAll('.tf-chip').forEach(chip => {
    chip.addEventListener('click', () => chip.classList.toggle('active'));
});

async function saveSettings() {
    const tfs = Array.from(document.querySelectorAll('.tf-chip.active')).map(c => c.dataset.tf);
    const data = {
        capital: parseFloat(document.getElementById('set-capital').value),
        risk_per_trade_pct: parseFloat(document.getElementById('set-risk').value),
        min_score: parseFloat(document.getElementById('set-score').value),
        analysis_interval_sec: parseInt(document.getElementById('set-interval').value),
        style_mode: document.getElementById('set-style').value,
        auto_generate_signals: document.getElementById('set-auto').checked,
        timeframes: tfs,
    };
    const res = await api('/api/settings', 'POST', data);
    if (res.ok) { alert('{{ t.save }}'); location.reload(); }
    else { alert('Error'); }
}
