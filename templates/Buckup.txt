let currentLang = localStorage.getItem("lang") || "ar";

function applyLanguage() {
    const isRtl = currentLang === "ar";

    document.documentElement.setAttribute("lang", currentLang);
    document.documentElement.setAttribute("dir", isRtl ? "rtl" : "ltr");

    const bs = document.getElementById("bootstrap-css");
    if (bs) {
        bs.href = isRtl
            ? "https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/css/bootstrap.rtl.min.css"
            : "https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/css/bootstrap.min.css";
    }

    const btnTxt = document.getElementById("lang-btn-text");
    if (btnTxt) btnTxt.innerText = isRtl ? "EN" : "AR";

    document.querySelectorAll("[data-i18n]").forEach(function (el) {
        const key = el.getAttribute("data-i18n");
        const dict = i18n[currentLang] || i18n.ar;
        if (dict[key]) el.innerText = dict[key];
    });
}

function toggleLanguage() {
    currentLang = currentLang === "ar" ? "en" : "ar";
    localStorage.setItem("lang", currentLang);
    applyLanguage();
}

async function api(url, method, body) {
    method = method || "GET";
    const opts = { method: method, headers: { "Content-Type": "application/json" } };
    if (body) opts.body = JSON.stringify(body);
    const r = await fetch(url, opts);
    return r.json();
}

async function approveSignal(uid) {
    if (!confirm(i18n[currentLang].confirm_execute)) return;
    const res = await api("/api/approve/" + uid, "POST");
    if (res.ok) {
        alert(i18n[currentLang].executed_success);
        location.reload();
    } else {
        alert(i18n[currentLang].rejected_msg + ": " + (res.error || ""));
    }
}

async function rejectSignal(uid) {
    await api("/api/reject/" + uid, "POST");
    location.reload();
}

async function runAnalysis() {
    const btn = event.target.closest("button");
    if (btn) btn.disabled = true;
    const res = await api("/api/run-analysis", "POST");
    alert(
        i18n[currentLang].analysis_done +
        "\n" + i18n[currentLang].generated + ": " + res.generated +
        "\n" + i18n[currentLang].rejected + ": " + res.rejected
    );
    location.reload();
}

function updateTimeframeChips() {
    const style = document.getElementById('set-style').value;
    const swingTfs = ['1H', '4H'];
    const scalpTfs = ['1M', '5M', '15M'];
    document.querySelectorAll('.tf-chip').forEach(chip => {
        const tf = chip.dataset.tf;
        let show = (style === 'swing' && swingTfs.includes(tf)) || 
                   (style === 'scalping' && scalpTfs.includes(tf)) || 
                   (style === 'both' && [...swingTfs, ...scalpTfs].includes(tf));
        chip.style.display = show ? 'inline-block' : 'none';
        if (!show) chip.classList.remove('active');
    });
}
document.getElementById('set-style').addEventListener('change', updateTimeframeChips);
document.addEventListener('DOMContentLoaded', updateTimeframeChips);

document.querySelectorAll('.tf-chip').forEach(chip => {
    chip.addEventListener('click', () => chip.classList.toggle('active'));
});

async function saveSettings() {
    const tfs = Array.from(document.querySelectorAll('.tf-chip.active')).map(c => c.dataset.tf);
    const data = {
        capital: parseFloat(document.getElementById('set-capital').value),
        risk_per_trade_pct: parseFloat(document.getElementById('set-risk').value),
        min_score: parseFloat(document.getElementById('set-score').value),
        analysis_interval_sec: parseInt(document.getElementById('set-interval').value),
        style_mode: document.getElementById('set-style').value,
        auto_generate_signals: document.getElementById('set-auto').checked,
        timeframes: tfs,
    };
    const res = await fetch('/api/settings', {
        method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(data)
    }).then(r => r.json());
    
    if (res.ok) { alert('تم الحفظ'); location.reload(); }
    else { alert('فشل'); }
}
